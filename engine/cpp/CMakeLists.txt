cmake_minimum_required(VERSION 3.16)
project(silencer_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(silencer_engine SHARED engine.cpp)
find_path(PCRE2_INCLUDE_DIR pcre2.h)
find_library(PCRE2_LIBRARY NAMES pcre2-8)
if(NOT PCRE2_INCLUDE_DIR OR NOT PCRE2_LIBRARY)
  message(FATAL_ERROR "PCRE2 development headers and libpcre2-8 are required")
endif()
target_include_directories(silencer_engine PRIVATE ${PCRE2_INCLUDE_DIR})
target_link_libraries(silencer_engine PRIVATE ${PCRE2_LIBRARY})
target_include_directories(silencer_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(silencer_engine PROPERTIES OUTPUT_NAME silencer_engine)
find_package(Threads REQUIRED)
add_executable(engine_test engine_test.cpp)
target_link_libraries(engine_test PRIVATE silencer_engine Threads::Threads)
include(CTest)
add_test(NAME engine_test COMMAND engine_test)

option(SILENCER_ENGINE_BUILD_BENCH "Build the offline C++ masking benchmark" OFF)
if(SILENCER_ENGINE_BUILD_BENCH)
  add_executable(engine_benchmark engine_benchmark.cpp)
  target_link_libraries(engine_benchmark PRIVATE silencer_engine)
endif()

option(SILENCER_ENGINE_BUILD_FUZZ "Build the libFuzzer engine harness" OFF)
if(SILENCER_ENGINE_BUILD_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "SILENCER_ENGINE_BUILD_FUZZ requires Clang and libFuzzer")
  endif()
  add_executable(engine_fuzz engine_fuzz.cpp)
  target_link_libraries(engine_fuzz PRIVATE silencer_engine)
  target_compile_options(engine_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(engine_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
